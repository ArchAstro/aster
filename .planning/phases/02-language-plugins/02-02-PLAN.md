2. Create src/targets/resolver.rs with TargetResolver:
   - Define standard target names as constants: "test", "build", "lint"
   - Define default commands per plugin:
     ```rust
     fn defaults_for_plugin(plugin_name: &str) -> HashMap<String, String> {
         match plugin_name {
             "nodejs" => hashmap! {
                 "test" => "npm test",
                 "build" => "npm run build",
                 "lint" => "npm run lint",
             },
             "elixir" => hashmap! {
                 "test" => "mix test",
                 "build" => "mix compile",
                 "lint" => "mix credo",
             },
             "python" => hashmap! {
                 "test" => "pytest",
                 "build" => "python -m build",
                 "lint" => "ruff check .",
             },
             _ => HashMap::new(),
         }
     }
     ```

3. Implement resolve() method:
   ```rust
   pub fn resolve(
       plugin_name: &str,
       custom_targets: &HashMap<String, String>,
   ) -> HashMap<String, String> {
       let mut targets = defaults_for_plugin(plugin_name);
       // Custom targets override defaults (merge at key level)
       for (name, command) in custom_targets {
           targets.insert(name.clone(), command.clone());
       }
       targets
   }
   ```

4. Add unit tests:
   - test_nodejs_defaults (returns npm commands)
   - test_elixir_defaults (returns mix commands)
   - test_python_defaults (returns pytest/ruff commands)
   - test_custom_override (custom "test" replaces default)
   - test_custom_addition (custom "deploy" added to defaults)
   - test_unknown_plugin (returns empty + custom only)
  </action>
  <verify>`cargo test targets` passes all target resolver tests</verify>
  <done>TargetResolver provides per-language default commands that can be overridden</done>
</task>
